  <script is:inline>
    // A single confetti burst for the moment the key appears — the one genuinely
    // good beat in a flow that otherwise reads as a spinner. Deliberately not a
    // library: the site promises no third-party scripts, and pulling a CDN
    // dependency onto the page that displays a licence key is a poor trade for a
    // decoration. ~40 lines of canvas costs nothing and ships nothing.
    function hostbeamConfetti() {
      // Honour reduced motion: for some people this is nausea, not delight.
      if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;

      const canvas = document.createElement("canvas");
      canvas.setAttribute("aria-hidden", "true");
      canvas.style.cssText =
        "position:fixed;inset:0;width:100%;height:100%;pointer-events:none;z-index:9999";
      document.body.appendChild(canvas);

      const ctx = canvas.getContext("2d");
      const dpr = Math.min(window.devicePixelRatio || 1, 2);
      canvas.width = innerWidth * dpr;
      canvas.height = innerHeight * dpr;
      ctx.scale(dpr, dpr);

      // Brand blues, with one warm accent so it reads as celebration not UI.
      const colors = ["#2d6cf6", "#5183f7", "#7aa2f8", "#a9c4fb", "#32d74b", "#ffc857"];
      const pieces = Array.from({ length: 90 }, () => ({
        x: innerWidth / 2 + (Math.random() - 0.5) * 120,
        y: innerHeight * 0.38,
        vx: (Math.random() - 0.5) * 9,
        vy: Math.random() * -11 - 4,
        w: 5 + Math.random() * 5,
        h: 8 + Math.random() * 6,
        rot: Math.random() * Math.PI,
        vr: (Math.random() - 0.5) * 0.3,
        color: colors[(Math.random() * colors.length) | 0],
      }));

      const started = performance.now();
      (function frame(now) {
        const elapsed = now - started;
        ctx.clearRect(0, 0, innerWidth, innerHeight);
        // Fade the whole burst out rather than letting pieces litter the floor.
        ctx.globalAlpha = Math.max(0, 1 - Math.max(0, elapsed - 1600) / 900);

        for (const p of pieces) {
          p.vy += 0.32;            // gravity
          p.vx *= 0.995;           // slight drag, so it drifts rather than flies
          p.x += p.vx;
          p.y += p.vy;
          p.rot += p.vr;
          ctx.save();
          ctx.translate(p.x, p.y);
          ctx.rotate(p.rot);
          ctx.fillStyle = p.color;
          ctx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
          ctx.restore();
        }

        if (elapsed < 2500) requestAnimationFrame(frame);
        else canvas.remove();
      })(started);
    }
  </script>

  <script is:inline define:vars={{ licenseApi: LICENSE_API }}>
    (async () => {
      const txn = new URLSearchParams(location.search).get("transaction_id");
      const show = (id) => {
        for (const s of ["waiting", "ready", "emailed"])
          document.getElementById(s).style.display = s === id ? "" : "none";
        // Only when the key is actually on screen. The "emailed" fallback means
        // something went slightly wrong; confetti there would read as sarcasm.
        if (id === "ready") hostbeamConfetti();
      };
      if (!txn) return show("emailed");

      // The webhook usually lands within a second or two of the redirect;
      // poll briefly before falling back to the email path.
      for (let attempt = 0; attempt < 15; attempt++) {
        try {
          const res = await fetch(`${licenseApi}/key?transaction_id=${encodeURIComponent(txn)}`);
          if (res.ok) {
            const { key } = await res.json();
            document.getElementById("key").textContent = key;
            const btn = document.getElementById("copy");
            btn.addEventListener("click", async () => {
              try {
                await navigator.clipboard.writeText(key);
                btn.textContent = "Copied";
                setTimeout(() => (btn.textContent = "Copy"), 1500);
              } catch {
                /* selection fallback: the key is visible and selectable */
              }
            });
            return show("ready");
          }
        } catch {
          /* transient network error — keep polling */
        }
        await new Promise((r) => setTimeout(r, 2000));
      }
      show("emailed");
    })();
  </script>
