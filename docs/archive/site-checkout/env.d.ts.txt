/// <reference types="astro/client" />

/** Build-time checkout configuration (see .env.example and src/lib/checkout.ts).
 * All PUBLIC_ values are inlined into the client bundle by Astro, so only
 * public identifiers belong here — never the Paddle API key. */
interface ImportMetaEnv {
  readonly PUBLIC_PADDLE_ENV?: "sandbox" | "production";
  readonly PUBLIC_PADDLE_CLIENT_TOKEN?: string;
  readonly PUBLIC_PADDLE_PRICE_ID?: string;
  readonly PUBLIC_LICENSE_API?: string;
  readonly PUBLIC_PADDLE_SANDBOX_HOSTS?: string;
}

interface ImportMeta {
  readonly env: ImportMetaEnv;
}
